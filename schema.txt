-- WARNING: This schema is for context only and is not meant to be run.
-- Table order and constraints may not be valid for execution.

CREATE TABLE public.members (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  name text,
  phone text,
  status USER-DEFINED,
  CONSTRAINT members_pkey PRIMARY KEY (id)
);
CREATE TABLE public.books (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  title text,
  author text,
  publisher text,
  isbn text UNIQUE,
  CONSTRAINT books_pkey PRIMARY KEY (id)
);
CREATE TABLE public.bookCopies (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  book_id bigint,
  copy_code text UNIQUE,
  status USER-DEFINED,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT bookCopies_pkey PRIMARY KEY (id),
  CONSTRAINT bookCopies_book_id_fkey FOREIGN KEY (book_id) REFERENCES public.books(id)
);
CREATE TABLE public.borrowings (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  member_id bigint,
  borrowed_at timestamp with time zone DEFAULT now(),
  due_date timestamp with time zone,
  status text,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  CONSTRAINT borrowings_pkey PRIMARY KEY (id),
  CONSTRAINT borrowings_member_id_fkey FOREIGN KEY (member_id) REFERENCES public.members(id)
);
CREATE TABLE public.borrowing_items (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  created_at timestamp with time zone NOT NULL DEFAULT now(),
  borrowing_id bigint,
  book_copy_id bigint,
  returned_at timestamp with time zone,
  status text,
  CONSTRAINT borrowing_items_pkey PRIMARY KEY (id),
  CONSTRAINT borrowing_items_borrowing_id_fkey FOREIGN KEY (borrowing_id) REFERENCES public.borrowings(id),
  CONSTRAINT borrowing_items_book_copy_id_fkey FOREIGN KEY (book_copy_id) REFERENCES public.bookCopies(id)
);
